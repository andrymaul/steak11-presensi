function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var result = [];
  
  if (data.length > 1) {
    for (var i = 1; i < data.length; i++) {
      result.push({
        waktuWIB: String(data[i][0] || ''),
        tipe: String(data[i][1] || 'Masuk'),
        outlet: String(data[i][2] || ''),
        nama: String(data[i][3] || ''),
        status: String(data[i][4] || 'Hadir'),
        ketWaktu: String(data[i][5] || '-'),
        fotoBase64: String(data[i][6] || ''),
        alamatGps: String(data[i][7] || '-'),
        catatan: String(data[i][8] || '')
      });
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var payload = JSON.parse(e.postData.contents);
    
    if (payload.action === 'save_theme') {
      var props = PropertiesService.getScriptProperties();
      props.setProperty('STEAK11_THEME', JSON.stringify(payload.theme));
      return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
    }

    if (payload.action === 'update') {
      var data = sheet.getDataRange().getValues();
      for (var i = 1; i < data.length; i++) {
        if (data[i][3] === payload.data.nama && data[i][0] === payload.data.waktuWIB) {
          sheet.getRange(i + 1, 2).setValue(payload.data.tipe);
          sheet.getRange(i + 1, 3).setValue(payload.data.outlet);
          sheet.getRange(i + 1, 5).setValue(payload.data.status);
          sheet.getRange(i + 1, 6).setValue(payload.data.ketWaktu);
          sheet.getRange(i + 1, 8).setValue(payload.data.alamatGps);
          sheet.getRange(i + 1, 9).setValue(payload.data.catatan);
          break;
        }
      }
    } else {
      sheet.appendRow([
        payload.waktuWIB || '',
        payload.tipe || 'Masuk',
        payload.outlet || '',
        payload.nama || '',
        payload.status || 'Hadir',
        payload.ketWaktu || '-',
        payload.fotoBase64 || '',
        payload.alamatGps || '-',
        payload.catatan || ''
      ]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
